// If you want to pass a key-value pair from outside via the command line of cpp use:
// cpp ... -D KEY=VALUE ... e.g. cpp -D KEY=1 ...

// or you can define it inside of the processed file:
#define TEST 1
#define IF1 1

//This is a comment - it will not appear in the output (if you need comments 
// in your output set -C Option of cpp)

This is common code! It will be shown in the output of the Pre-Processor!

// use #ifdef for conditional parts
#ifdef TEST
"This is shown if TEST was set"
#else
"This is shown if TEST was **NOT** set"
#endif

// "defined" yields true if the variable is set to any value
#if defined IF1 || defined IF2 
  "This is shown if IF1 OR IF2 were set" 
#endif 

// You may use logical || and &&
#if defined IF1 && defined IF2 
  "This is shown if IF1 AND IF2 were set" 
#else
 //text in quotes is NOT substituted!
 "IF1 OR IF2 is not substituted by its value! because of surrounding quotes"
#endif 

#define IF2 0

#if defined IF1 && defined IF2 
  "This is shown if IF1 AND IF2 are set (even to the value 0)" 
#endif 

#if IF1==1
  "IF1 has value 1"
#endif

// #IF does only support comparing constant numbers
// so we have to tweak it to make the code more readable:
// insert a named variable representing a number and compare to that 'value'

#define OPTION1 1
#define OPTION0 0

#if IF2==OPTION0
  "IF2 has value OPTION0 - which is 0 because of the define"
#endif

 #define make_string(s) str(s)

#define FAST_FORWARD_KEY >

// Multi line defines ************************************
// you want to assign several lines of text by a single define?
// You use \ as a line continuation escape character.
// Attention! The \ character must be the last character on the line. 
// If it is not (even if it is just white space afterward) you will get confusing error messages on each line after it.

#define MULTI_LINE_DEFINE { \
                       (a) ^= (b); \
                       (b) ^= (a); \
                       (a) ^= (b); \
                   }
